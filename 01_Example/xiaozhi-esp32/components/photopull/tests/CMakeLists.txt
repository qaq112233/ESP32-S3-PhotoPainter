cmake_minimum_required(VERSION 3.16)
project(photopull_host_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PHOTOPULL_SANITIZERS "Run host tests with address and undefined behavior sanitizers" ON)

function(photopull_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ..)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(PHOTOPULL_SANITIZERS)
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
